# EXPERIMENT 8 - LINEAR REGRESSION
# ============================================================

import numpy as np
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error, r2_score

# ------------------------------------------------------------
# 1. Create Dataset
# ------------------------------------------------------------

# Input: Hours studied
X = np.array([[1], [2], [3], [4], [5]])

# Output: Marks obtained
y = np.array([20, 30, 40, 50, 60])

print("TRAINING DATA")
print("-" * 30)
print("Hours Studied | Marks")

for hours, marks in zip(X.flatten(), y):
    print(f"{hours:5d}         | {marks}")

# ------------------------------------------------------------
# 2. Train Linear Regression Model
# ------------------------------------------------------------

model = LinearRegression()
model.fit(X, y)

# ------------------------------------------------------------
# 3. Display Model Parameters
# ------------------------------------------------------------

slope = model.coef_[0]
intercept = model.intercept_

print("\nREGRESSION MODEL")
print("Slope:", round(slope, 2))
print("Intercept:", round(intercept, 2))

print(
    f"Equation: Marks = {slope:.2f} * Hours "
    f"+ {intercept:.2f}"
)

# ------------------------------------------------------------
# 4. Predict Marks for a New Input
# ------------------------------------------------------------

new_hours = np.array([[6]])
predicted_marks = model.predict(new_hours)

print("\nPREDICTION")
print("Hours Studied:", new_hours[0, 0])
print("Predicted Marks:", round(predicted_marks[0], 2))

# ------------------------------------------------------------
# 5. Evaluate Model
# ------------------------------------------------------------

y_pred = model.predict(X)

mse = mean_squared_error(y, y_pred)
r2 = r2_score(y, y_pred)

print("\nMODEL EVALUATION")
print("Mean Squared Error:", round(mse, 2))
print("R-squared Score:", round(r2, 2))

# ------------------------------------------------------------
# 6. Plot Regression Line
# ------------------------------------------------------------

X_line = np.linspace(0, 7, 100).reshape(-1, 1)
y_line = model.predict(X_line)

plt.figure(figsize=(9, 6))

plt.scatter(
    X, y,
    color="blue",
    s=90,
    label="Actual Data"
)

plt.plot(
    X_line, y_line,
    color="red",
    linewidth=2.5,
    label="Regression Line"
)

plt.scatter(
    new_hours, predicted_marks,
    color="green",
    marker="*",
    s=250,
    label="Predicted Value"
)

plt.annotate(
    f"Prediction = {predicted_marks[0]:.0f}",
    (new_hours[0, 0], predicted_marks[0]),
    xytext=(10, -20),
    textcoords="offset points"
)

plt.xlabel("Hours Studied")
plt.ylabel("Marks Obtained")
plt.title("Linear Regression: Hours Studied vs Marks")
plt.legend()
plt.grid(True, linestyle="--", alpha=0.5)
plt.tight_layout()
plt.show()

# ------------------------------------------------------------
# 7. Result
# ------------------------------------------------------------

print("\nRESULT:")
print("Linear Regression implemented successfully.")
print("Predicted marks for 6 hours:", predicted_marks[0])
