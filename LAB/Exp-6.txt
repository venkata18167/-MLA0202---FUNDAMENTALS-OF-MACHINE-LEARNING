import matplotlib.pyplot as plt
import numpy as np

from sklearn.datasets import load_iris
from sklearn.model_selection import train_test_split
from sklearn.naive_bayes import GaussianNB
from sklearn.metrics import confusion_matrix, accuracy_score

# ------------------------------------------------------------
# 1. Load Dataset
# ------------------------------------------------------------

iris = load_iris()

X = iris.data
y = iris.target

print("IRIS DATASET")
print("-" * 40)
print("Total samples:", len(X))
print("Number of features:", X.shape[1])
print("Flower classes:", list(iris.target_names))

# ------------------------------------------------------------
# 2. Split Dataset into Training and Testing Sets
# ------------------------------------------------------------

X_train, X_test, y_train, y_test = train_test_split(
    X,
    y,
    test_size=0.30,
    random_state=42,
    stratify=y
)

print("\nTraining samples:", len(X_train))
print("Testing samples:", len(X_test))

# ------------------------------------------------------------
# 3. Create and Train Gaussian Naive Bayes Model
# ------------------------------------------------------------

model = GaussianNB()
model.fit(X_train, y_train)

# ------------------------------------------------------------
# 4. Predict Test Data
# ------------------------------------------------------------

y_pred = model.predict(X_test)

print("\nPREDICTED CLASSES")
print(y_pred)

print("\nACTUAL CLASSES")
print(y_test)

# ------------------------------------------------------------
# 5. Calculate Confusion Matrix and Accuracy
# ------------------------------------------------------------

cm = confusion_matrix(
    y_test,
    y_pred,
    labels=[0, 1, 2]
)

accuracy = accuracy_score(y_test, y_pred)

print("\nCONFUSION MATRIX")
print(cm)

print("\nCLASSIFICATION ACCURACY")
print(f"{accuracy:.4f} ({accuracy * 100:.2f}%)")

# ------------------------------------------------------------
# 6. Plot Confusion Matrix
# ------------------------------------------------------------

fig, ax = plt.subplots(figsize=(7, 5))

image = ax.imshow(cm, cmap="Blues")

ax.set_title(
    "Naive Bayes - Confusion Matrix",
    fontsize=14,
    fontweight="bold"
)

ax.set_xlabel("Predicted Class")
ax.set_ylabel("Actual Class")

ax.set_xticks(range(3))
ax.set_yticks(range(3))

ax.set_xticklabels(iris.target_names)
ax.set_yticklabels(iris.target_names)

# Display the values inside the cells
threshold = cm.max() / 2

for i in range(cm.shape[0]):
    for j in range(cm.shape[1]):
        ax.text(
            j, i, str(cm[i, j]),
            ha="center",
            va="center",
            color="white" if cm[i, j] > threshold else "black",
            fontsize=13
        )

plt.colorbar(image, ax=ax, label="Number of Samples")
plt.tight_layout()
plt.show()

# ------------------------------------------------------------
# 7. Result
# ------------------------------------------------------------

print("\nRESULT:")
print("Naive Bayes classification completed successfully.")
print(f"Accuracy obtained: {accuracy * 100:.2f}%")
